<figure class="carousel" aria-roledescription="carousel" aria-label={ariaLabel}>
  <div class="carousel-track" tabindex="0">
    {items.map((item, i) => (
      <div class="carousel-slide" data-slide-index={i} style={`scroll-snap-align: start;`}>
        <div class="slide-frame" style={`aspect-ratio: ${item.aspect ?? "3 / 2"};`}>
          <img src={item.src} alt={item.caption ?? ""} loading="lazy" decoding="async" />
        </div>
        {item.caption && <figcaption class="slide-caption">{item.caption}</figcaption>}
      </div>
    ))}
  </div>

  {items.length > 1 && (
    <div class="carousel-controls">
      <button type="button" class="carousel-btn carousel-prev" aria-label="Previous photo">‹</button>
      <ol class="carousel-dots" role="tablist">
        {items.map((_, i) => (
          <li>
            <button class="carousel-dot" data-dot-index={i} aria-label={`Go to photo ${i+1}`}></button>
          </li>
        ))}
      </ol>
      <button type="button" class="carousel-btn carousel-next" aria-label="Next photo">›</button>
    </div>
  )}
</figure>

<script>
  const { items } = Astro.props;
  const carousel = document.querySelector('.carousel');
  if (carousel.dataset.bound) return;
  carousel.dataset.bound = 'true';

  const track = carousel.querySelector('.carousel-track');
  const slides = Array.from(carousel.querySelectorAll('.carousel-slide'));
  const dots = Array.from(carousel.querySelectorAll('.carousel-dot'));
  const prevBtn = carousel.querySelector('.carousel-prev');
  const nextBtn = carousel.querySelector('.carousel-next');

  let slideWidth = 0;
  let currentSlide = 0;
  let observer = null;

  const updateSlideWidth = () => {
    if (slides.length > 0) {
      slideWidth = slides[0].offsetWidth;
    }
  };

  const updateCurrentSlide = (index) => {
    currentSlide = index;
    dots.forEach((dot, i) => {
      dot.setAttribute('aria-current', i === index ? 'true' : 'false');
    });
    prevBtn.disabled = index === 0;
    nextBtn.disabled = index === slides.length - 1;
  };

  const scrollToSlide = (index) => {
    const slide = slides[index];
    if (!slide) return;

    const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    const behavior = prefersReducedMotion ? 'auto' : 'smooth';

    slide.scrollIntoView({ behavior, inline: 'start', block: 'nearest' });
  };

  const handlePrev = () => {
    if (currentSlide > 0) {
      scrollToSlide(currentSlide - 1);
    }
  };

  const handleNext = () => {
    if (currentSlide < slides.length - 1) {
      scrollToSlide(currentSlide + 1);
    }
  };

  const handleDotClick = (e) => {
    const index = parseInt(e.target.dataset.dotIndex);
    scrollToSlide(index);
  };

  const handleKeyDown = (e) => {
    if (!track.matches(':focus-within')) return;

    switch (e.key) {
      case 'ArrowLeft':
        handlePrev();
        break;
      case 'ArrowRight':
        handleNext();
        break;
      case 'Home':
        scrollToSlide(0);
        break;
      case 'End':
        scrollToSlide(slides.length - 1);
        break;
    }
  };

  const setupIntersectionObserver = () => {
    const threshold = 0.6;
    observer = new IntersectionObserver((entries) => {
      entries.forEach((entry) => {
        if (entry.isIntersecting) {
          const index = parseInt(entry.target.dataset.slideIndex);
          updateCurrentSlide(index);
        }
      });
    }, { threshold });
    slides.forEach((slide) => observer.observe(slide));
  };

  const init = () => {
    updateSlideWidth();
    setupIntersectionObserver();
    updateCurrentSlide(0);

    prevBtn.addEventListener('click', handlePrev);
    nextBtn.addEventListener('click', handleNext);
    dots.forEach((dot) => dot.addEventListener('click', handleDotClick));
    track.addEventListener('keydown', handleKeyDown);

    window.addEventListener('resize', () => {
      updateSlideWidth();
    });
  };

  document.addEventListener('astro:after-swap', init);
  init();
</script>
