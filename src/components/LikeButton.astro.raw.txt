const { slug } = Astro.props as Props;

<button class="like-button" data-slug={slug} aria-label="Like this post">
  <svg class="heart" width="24" height="24" viewBox="0 0 24 24" fill="none" xmlns="http://www.w3.org/2000/svg">
    <path class="heart-path" d="M12 21.35L10.55 20.03C5.4 15.36 2 12.28 2 8.5 2 5.42 4.42 3 7.5 3C9.24 3 10.91 3.81 12 5 13.09 3.81 14.76 3 16.5 3 19.58 3 22 5.42 22 8.5C22 12.28 18.6 15.36 13.45 20.03L12 21.35Z" stroke="currentColor" stroke-width="1.5"/>
  </svg>
  <span class="like-count">0</span>
</button>

<script>
  const button = document.currentScript?.previousElementSibling;
  if (!button || !(button instanceof HTMLButtonElement)) {
    throw new Error('Unexpected button element');
  }

  const slug = button.dataset.slug;
  if (!slug) {
    throw new Error('Missing data-slug attribute');
  }

  const countElement = button.querySelector('.like-count');
  const heartPath = button.querySelector('.heart-path');

  if (!countElement || !heartPath) {
    throw new Error('Missing count or heart element');
  }

  const storageKey = `like-${slug}`;

  function updateCount(count) {
    countElement.textContent = count > 99 ? '99+' : count.toString();
    if (count > 0) {
      button.classList.add('is-liked');
    } else {
      button.classList.remove('is-liked');
    }
  }

  function loadCount() {
    const stored = localStorage.getItem(storageKey);
    const count = stored ? parseInt(stored, 10) : 0;
    updateCount(count);
  }

  function saveCount(count) {
    localStorage.setItem(storageKey, count.toString());
    updateCount(count);
  }

  function handleClick() {
    const currentCount = parseInt(countElement.textContent, 10) || 0;
    const newCount = currentCount + 1;
    saveCount(newCount);
    button.classList.add('just-clicked');
    setTimeout(() => {
      button.classList.remove('just-clicked');
    }, 380);
  }

  button.addEventListener('click', handleClick);
  loadCount();
</script>
