import Layout from "../../layouts/Layout.astro";
import { getCollection, render } from "astro:content";
import type { CollectionEntry } from "astro:content";

export async function getStaticPaths() {
  const entries = await getCollection("work");
  return entries.map((entry) => ({
    params: { slug: entry.id },
    props: { entry },
  }));
}

type Props = { entry: CollectionEntry<"work"> };
const { entry } = Astro.props;
const { Content } = await render(entry);

<Layout title={`${entry.data.company} — Alok Kumar`} description={entry.data.summary}>
  <div class="container-content">
    <div class="py-12 md:py-20">
      <div class="eyebrow">Work — {entry.data.period}</div>
      <h1 class="display mt-4 mb-2">{entry.data.company}</h1>
      <div class="text-sans ink-muted mt-2 mb-6">{entry.data.role}</div>
      <p class="lede font-serif text-2xl font-light max-w-40ch">{entry.data.summary}</p>
      <div class="flex flex-wrap gap-4 mt-8">
        <a href={entry.data.website} target="_blank" class="link link-accent">Visit {entry.data.company} ↗</a>
        <a href="/work" class="link link-accent">Back to work</a>
      </div>
    </div>
  </div>

  <div class="container-prose prose prose-lg">
    <Content />
  </div>

  {entry.data.projects.length > 0 && (
    <div class="container-content py-12 md:py-20">
      <div class="eyebrow">Selected projects</div>
      <div class="container-prose prose prose-lg mt-6">
        {entry.data.projects.map((project, index) => (
          <div key={index} class="mt-12">
            {index > 0 && <hr class="my-12" />}
            <h2 class="text-2xl font-normal font-serif mt-0 mb-4">{project.title}</h2>
            <p class="text-sans ink-muted text-pretty max-w-55ch">{project.description}</p>
            {project.href && (
              <a href={project.href} class="link link-accent mt-2 block">Read more ↗</a>
            )}
          </div>
        ))}
      </div>
    </div>
  )}

  {entry.data.testimonial && (
    <div class="container-content py-12 md:py-20 text-center">
      <blockquote class="display-italic text-3xl leading-tight text-balance max-w-30ch mx-auto">
        {entry.data.testimonial.quote}
      </blockquote>
      <p class="text-sans ink-muted text-sm mt-4">
        — {entry.data.testimonial.author}{entry.data.testimonial.role && `, ${entry.data.testimonial.role}`}
      </p>
    </div>
  )}

  <div class="container-content py-8 md:py-12">
    <hr />
    <div class="flex flex-wrap items-center justify-between gap-4 mt-8">
      <div class="eyebrow">Next</div>
      <a href="/work" class="link link-accent font-serif">Back to work →</a>
    </div>
  </div>
</Layout>
